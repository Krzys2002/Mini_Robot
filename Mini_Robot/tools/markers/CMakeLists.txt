# Offline tooling only. No detector or OpenCV dependency is added to domain targets.
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
if(OpenCV_VERSION VERSION_LESS 4.2)
    message(FATAL_ERROR "AprilTag marker tools require OpenCV 4.2 or newer")
endif()
if(OpenCV_VERSION VERSION_LESS 4.7)
    find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs aruco)
else()
    find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs objdetect)
endif()
add_library(micro_robot_marker_print STATIC MarkerPrint.cpp MarkerPrint.hpp)
target_include_directories(micro_robot_marker_print PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(micro_robot_marker_print SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
target_link_libraries(micro_robot_marker_print PUBLIC ${OpenCV_LIBS})
micro_robot_target_defaults(micro_robot_marker_print)
add_executable(micro_robot_print_markers main.cpp)
target_link_libraries(micro_robot_print_markers PRIVATE micro_robot_marker_print)
micro_robot_target_defaults(micro_robot_print_markers)
